        function formatMinutes(totalMinutes) {
            const hours = Math.floor(totalMinutes / 60);
            const minutes = totalMinutes % 60;
            return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
        }

        function updateTotal(id) {
            const startVal = parseInt(document.getElementById(`start-${id}`).value);
            const endVal = parseInt(document.getElementById(`end-${id}`).value);
            
            let diff = endVal - startVal;
            if (diff < 0) diff = 0; // Prevent negative time if end is before start
            
            const hours = diff / 60;
            document.getElementById(`total-${id}`).textContent = `${hours.toFixed(1)} t`;
        }

        function updateTime(slider, outputId) {
            const output = document.getElementById(outputId);
            output.textContent = formatMinutes(slider.value);
            
            // Extract the id number to update the specific total
            const idMatch = slider.id.match(/\d+$/);
            if (idMatch) {
                updateTotal(idMatch[0]);
            }
        }
    </script>
